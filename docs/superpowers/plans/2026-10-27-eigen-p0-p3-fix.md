`framework/sdc_numa.cpp`（完整）：
```cpp
/* SPDX-License-Identifier: Apache-2.0 */
#include "sdc_numa.h"
#if defined(__aarch64__) && defined(__linux__)

#include <dirent.h>
#include <cstdio>
#include <cstring>

/* sysfs 目录项名过滤: nodeN 且 N 全数字 */
static int parse_node_dir(const char *name, int *out)
{
    if (strncmp(name, "node", 4) != 0) return 0;
    char *end;
    long n = strtol(name + 4, &end, 10);
    if (*end != '\0' || n < 0) return 0;
    *out = (int)n;
    return 1;
}

/* 读 /sys/devices/system/node/nodeN/<field> 的首个数值(kB), 失败 -1 */
static long read_node_kb(int node, const char *field)
{
    char path[128], buf[256];
    snprintf(path, sizeof path, "/sys/devices/system/node/node%d/%s", node, field);
    FILE *f = fopen(path, "r");
    if (!f) return -1;
    long v = -1;
    if (fscanf(f, "%ld", &v) != 1) v = -1;
    fclose(f);
    return v;
}

const struct sdc_numa_topology *sdc_numa_probe(void)
{
    /* (PR#302 检视 #2 勘误: 原草图用 static bool done 普通标志——并发
     * 首调可读到半初始化 topo 且构成数据竞争。实现采用 C++11 magic
     * static(函数局部 const 静态初始化, 语言保证线程安全一次性),
     * 本例同步为实现形态: static const struct sdc_numa_topology topo
     * = probe_uncached(); 全部探测逻辑移入 probe_uncached()。) */
    static sdc_numa_topology topo;
    static bool done = false;
    if (!done) {
        done = true;
        topo.total_cpus = 0;
        topo.nodes_with_mem = 0;
        DIR *d = opendir("/sys/devices/system/node");
        if (d) {
            struct dirent *de;
            while ((de = readdir(d)) != nullptr) {
                int node;
                if (!parse_node_dir(de->d_name, &node)) continue;
                long cpus = read_node_kb(node, "cpulist");  /* 仅探测存在性 */
                (void)cpus;
                long total = read_node_kb(node, "meminfo"); /* meminfo 首行是 MemTotal */
                /* meminfo 是多行文件, read_node_kb 只拿首字段——改走整读: */
                char path[128];
                snprintf(path, sizeof path,
                         "/sys/devices/system/node/node%d/meminfo", node);
                FILE *f = fopen(path, "r");
                long memtotal_kb = -1;
                if (f) {
                    char line[256];
                    while (fgets(line, sizeof line, f)) {
                        if (sscanf(line, " Node %*d MemTotal: %ld kB", &memtotal_kb) == 1
                                || strstr(line, "MemTotal:"))
                            { if (memtotal_kb < 0)
                                  sscanf(strstr(line, ":") + 1, "%ld", &memtotal_kb);
                              break; }
                    }
                    fclose(f);
                }
                if (memtotal_kb > 0)
                    ++topo.nodes_with_mem;
            }
            closedir(d);
        }
        FILE *onl = fopen("/sys/devices/system/cpu/online", "r");
        if (onl) { /* 计在线 CPU 数(区间和)。(PR#302 检视 #3 勘误: 原草图
             * 的 sscanf+strchr(p,'-') 对 "0,2-3" 型输入有缺陷——首
             * token "0" 只匹配 1 个转换, b 未被写; strchr 在剩余串里
             * 找到后续 token 的 '-', b 保持未定值。实现的
             * for_each_cpu_in_list() 是正确形态: 先 b=a, 仅当数字后
             * 紧邻 '-' 才解析区间上界, 遇 ',' 才进下一 token——下方
             * 以实现形态重写本例。) */
            char buf[256] = "";
            if (fgets(buf, sizeof buf, onl)) {
                int n = 0; const char *p = buf;
                while (*p) {
                    char *end;
                    long a = strtol(p, &end, 10);
                    if (end == p) break;
                    long b = a;
                    if (*end == '-') b = strtol(end + 1, &end, 10);
                    if (b < a) b = a;
                    n += (int)(b - a + 1);
                    if (*end != ',') break;
                    p = end + 1;
                }
                topo.total_cpus = n;
            }
            fclose(onl);
        }
    }
    return &topo;
}

double sdc_ar_factor(enum sdc_mem_family fam, const struct sdc_numa_topology *t)
{
    if (!t) t = sdc_numa_probe();
    /* 分档表(spec §4.1): 行=有内存节点数, 列=负载族。
     * 探测失败(nodes_with_mem==0)与未知档一律最差档。 */
    static const struct { int nodes; double mem, comp; } tbl[] = {
        { 1, 25.0, 5.0 },   /* 单节点内存: SVD 实测带 18-25 取上界; GEMM 4.4+余量 */
        { 2, 10.0, 4.0 },   /* 保守估计, TODO(实测): 无真 2 节点内存机器 */
        { 4,  5.0, 3.0 },   /* 同上 */
        { 8,  4.0, 2.5 },   /* >=8 节点 */
    };
    int n = t->nodes_with_mem;
    double worst = fam == SDC_FAM_MEM_BOUND ? 25.0 : 5.0;
    if (n <= 0) return worst;
    for (const auto &r : tbl)
        if (n <= r.nodes) return fam == SDC_FAM_MEM_BOUND ? r.mem : r.comp;
    return fam == SDC_FAM_MEM_BOUND ? 4.0 : 2.5;
}

int sdc_numa_node_of_cpu(int cpu)
{
    if (cpu < 0) return -1;
    DIR *d = opendir("/sys/devices/system/node");
    if (!d) return -1;
    struct dirent *de;
    int found = -1;
    while ((de = readdir(d)) != nullptr) {
        int node;
        if (!parse_node_dir(de->d_name, &node)) continue;
        char path[128], buf[1024];
        snprintf(path, sizeof path, "/sys/devices/system/node/node%d/cpulist", node);
        FILE *f = fopen(path, "r");
        if (!f) continue;
        if (fgets(buf, sizeof buf, f)) {
            int a, b; char *p = buf;
            while (sscanf(p, "%d-%d", &a, &b) == 2 || sscanf(p, "%d", &a) == 1) {
                b = (strchr(p, '-') ? b : a);
                if (cpu >= a && cpu <= b) { found = node; break; }
                char *comma = strchr(p, ',');
                if (!comma) break;
                p = comma + 1;
            }
        }
        fclose(f);
        if (found >= 0) break;
    }
    closedir(d);
    return found;
}

#endif /* __aarch64__ && __linux__ */
```

```c
/* [可用内存预算] 三源取 min(spec §4.2): 物理×‰ / MemAvailable×0.9 /
 * cgroup 限额−已用。任一来源读失败(返回 0)即跳过该来源。
 * 动机: 旧口径只读 _SC_PHYS_PAGES, 批处理/容器限额下超分——
 * "job oom" 实测前科(svd_cdouble_sve.cpp:114-120 注释)。 */
static inline uint64_t sdc_memavailable_bytes(void)
{
    FILE *f = fopen("/proc/meminfo", "r");
    if (!f) return 0;
    char line[256];
    uint64_t kb = 0;
    while (fgets(line, sizeof line, f))
        if (sscanf(line, "MemAvailable: %lu kB", &kb) == 1) break;
    fclose(f);
    return kb * 1024;
}

static inline uint64_t sdc_cgroup_limit_bytes(void)
{
    /* v2: memory.max - memory.current ("max" = 无限 → 0)
     * v1: memory/memory.limit_in_bytes - memory.usage_in_bytes
     *      (limit >= 物理内存视为无限 → 0)。本机(openEuler)实测为 v1。 */
    uint64_t lim, cur;
    FILE *f = fopen("/sys/fs/cgroup/memory.max", "r");           /* v2 */
    if (f) {
        char buf[64];
        bool ok = fgets(buf, sizeof buf, f) != nullptr;
        fclose(f);
        if (ok && strcmp(buf, "max\n") != 0) {
            lim = strtoull(buf, nullptr, 10);
            f = fopen("/sys/fs/cgroup/memory.current", "r");
            if (!f) return 0;
            ok = fscanf(f, "%lu", &cur) == 1 && lim > cur;
            fclose(f);
            return ok ? lim - cur : 0;
        }
    }
    f = fopen("/sys/fs/cgroup/memory/memory.limit_in_bytes", "r"); /* v1 */
    if (!f) return 0;
    bool ok = fscanf(f, "%lu", &lim) == 1;
    fclose(f);
    if (!ok || lim >= sdc_phys_mem_bytes()) return 0;   /* 无限 */
    f = fopen("/sys/fs/cgroup/memory/memory.usage_in_bytes", "r");
    if (!f) return 0;
    ok = fscanf(f, "%lu", &cur) == 1 && lim > cur;
    fclose(f);
    return ok ? lim - cur : 0;
}

static inline uint64_t sdc_budget_bytes(int permille)
{
    uint64_t b = sdc_phys_mem_bytes() * (uint64_t)permille / 1000;
    uint64_t ma = sdc_memavailable_bytes();
    if (ma > 0 && ma * 9 / 10 < b) b = ma * 9 / 10;
    uint64_t cg = sdc_cgroup_limit_bytes();
    if (cg > 0 && cg < b) b = cg;
    return b;
}
```
